# --- Cliente HTTP ------------------------------------------------------------
# `requests` lo usa nucleo.py (sesion con reintentos) y directorio.py.
requests==2.34.2
# urllib3 se usa de forma explicita: nucleo.py importa Retry para la politica
# de reintentos, asi que se fija aunque tambien sea dependencia de requests.
urllib3==2.7.0
# `httpx` lo usan red.py (descargas concurrentes), fuentes_indeed.py,
# fuentes_cursos.py, cursos.py y medir_verificacion.py.
httpx==0.28.1

# --- Parseo de HTML ----------------------------------------------------------
# BeautifulSoup lo usan excavadora.py, fuentes_linkedin.py, fuentes_mexico.py
# y fuentes_ats_extra.py.
beautifulsoup4==4.15.0
# lxml NO es opcional aunque no se importe por su nombre: esos modulos llaman a
# BeautifulSoup(html, "lxml"). Sin lxml, bs4 lanza FeatureNotFound y esas
# fuentes devuelven cero sin un error visible — un fallo silencioso.
lxml==6.1.3
